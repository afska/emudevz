`LDA #$01`    ; Загружает число $01 в регистр [A]
`LDA $01`     ; Загружает значение по адресу $0001 в [A]
`LDA $1F4B`   ; Загружает значение по адресу $1F4B в [A]
`STA $4070`   ; Записывает [A] по адресу $4070
`STA $60,X`   ; Записывает [A] по адресу $60+[X]
`STA $2030,X` ; Записывает [A] по адресу $2030+[X]
`STA ($01,X)` ; = STA ($01+X)
`STA ($03),Y` ; = STA ($03)+Y
  ; Варианты для [X] и [Y]: `LDX`, `STX`, `LDY`, `STY`
`TXA`         ; Переносит [X] в [A]
  ; Варианты: `TAX`, `TYA`, `TAY`
`ADC #30`     ; Прибавляет 30 к [A] (с учётом флага переноса)
`SBC #30`     ; Вычитает 30 из [A] (с учётом флага переноса)
`INX`         ; Увеличивает [X] на единицу
  ; Вариант для [Y]: `INY`
`DEX`         ; Уменьшает [X] на единицу
  ; Вариант для [Y]: `DEY`
@jump:        ; Определяет метку, к которой можно обратиться позже
`CPX #3`      ; Устанавливает Z=1, если [X] == 3
  ; Вариант для [Y]: `CPY`
`BEQ @jump`   ; Переходит к @jump, если Z == 1
`BNE @jump`   ; Переходит к @jump, если Z == 0
`JMP $4030`   ; Переходит к $4030 (устанавливает [PC] = $4030)
`JMP ($4030)` ; Переходит по адресу, составленному из байтов в $4030 и $4031
`PHA`         ; Помещает [A] в стек
`PLA`         ; Извлекает значение из стека в [A]
`JSR $4029`   ; Переходит к подпрограмме по адресу $4029
`RTS`         ; Возвращается к предыдущей инструкции
`BRK`         ; Завершает программу
